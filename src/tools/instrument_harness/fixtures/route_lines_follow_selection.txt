# route_lines_follow_selection.txt - the ROUTELINES overlay must follow
# the modulation line's selection.
#
# The lines used to be anchored to a source's ROUTE button in the
# modulation list, and were drawn for whichever source that button
# belonged to. The list is gone; the lines are anchored to the
# modulation LINE now, and they follow the modulator the line is
# showing.
#
# So the line's LEFT/RIGHT cycles the connected modulators, and the
# overlay's source follows.

FRAMES 4

# --- The line is the overlay's anchor -------------------------------------
JUMP MODLINE
FRAMES 2
ASSERT selected==MODLINE
ASSERT routelinesrc==0

# --- RIGHT cycles the line, and the overlay follows -----------------------
RIGHT
FRAMES 2
ASSERT modline_index==1
ASSERT routelinesrc==1

RIGHT
FRAMES 2
ASSERT modline_index==2
ASSERT routelinesrc==2

# --- LEFT cycles back -----------------------------------------------------
LEFT
FRAMES 2
ASSERT modline_index==1
ASSERT routelinesrc==1

# --- The overlay follows the line all the way round ----------------------
LEFT
FRAMES 2
ASSERT modline_index==0
ASSERT routelinesrc==0
RIGHT
FRAMES 2
RIGHT
FRAMES 2
RIGHT
FRAMES 2
# The line wraps round to the first connected modulator.
ASSERT modline_index==0
ASSERT routelinesrc==0
