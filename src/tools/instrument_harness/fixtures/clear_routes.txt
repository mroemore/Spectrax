# clear_routes.txt - the modulation screen's CLEAR ALL: the route list's
# last button opens the clear-all confirm (NO default). EDIT on NO
# cancels; RIGHT to YES + EDIT clears every route of the source.
#
# The instrument screen's modulation list is gone -- the clear-all moved
# to the modulation screen with the rest of the interface.
#
# The route-removal semantics are asserted at the unit level
# (test_modsystem's removeModulationsForSource tests); this fixture
# covers the UI flow.

FRAMES 4

# --- Open the modulation screen -------------------------------------------
JUMP MODLINE
EDIT
FRAMES 2
ASSERT topLayerSelected==MODSRC_0

# --- Route the selected source into a destination -------------------------
# "+ ROUTE" opens the destination picker; picking one routes the source
# into it.
JUMPTL MOD_ADDRUTE
FRAMES 1
EDIT
FRAMES 2
ASSERT topLayerSelected==DEST_0
EDIT
FRAMES 2
ASSERT topLayerSelected==MODSRC_0

# --- CLEAR ALL opens the confirm (NO default) -----------------------------
# The route list's last button.
JUMPTL MOD_CLEARALL
FRAMES 1
ASSERT topLayerSelected==MOD_CLEARALL
EDIT
FRAMES 2
ASSERT topLayerSelected==CLEARALL_NO

# --- EDIT again on NO cancels: the confirm closes -------------------------
# The screen's own selection is where it was -- the confirm was opened
# from the CLEAR ALL button.
EDIT
FRAMES 2
ASSERT topLayerSelected==MOD_CLEARALL

# --- Re-open and confirm YES clears every route of the source -----------
JUMPTL MOD_CLEARALL
FRAMES 1
EDIT
FRAMES 2
ASSERT topLayerSelected==CLEARALL_NO
RIGHT
ASSERT topLayerSelected==CLEARALL_YES
EDIT
FRAMES 2
# The screen comes back with an empty route list.
JUMPTL MOD_CLEARALL
FRAMES 1
ASSERT topLayerSelected==MOD_CLEARALL
