# chip_meta.txt — Task 7 fixture (Task 5 update)
#
# Walks the arranger chip row (chip 0) through expand -> swatch ->
# label edit -> collapse -> jump to instrument page, then exercises
# the meta row (TNEXT cycles voice type, VOICES dial adjusts the
# count) on the same chip's instrument screen.
#
# Task 5 rework notes:
#   * The arranger grid is now a scrollable window of per-cell graph
#     nodes. The single legacy ArrangerGuiNode grid is GONE. The
#     initial selected cell column is whatever s1.sng persisted in
#     selectedArrangerCell[0] (last-run state). We LEFT x5 first to
#     drive selected_x to 0 so the subsequent UP lands on chip 0
#     (column-matched nav from the windowed grid).
#   * The instrument meta row is now [TPREV] [TYPE-tag label
#     (non-selectable)] [TNEXT] [VOICES] [WIDTH] — the old single
#     cycling TYPE button is gone. JUMP VOICES + LEFT now lands on
#     TNEXT (the next-type action button), and EDIT on TNEXT fires
#     cbTypeNext, cycling voice type FM -> BLEP -> SAMPLE.

# Boot is SCENE_INSTRUMENT (per the harness's startup override);
# switch to SCENE_ARRANGER (user-facing 0) to access the chip row.
SCENE 0
FRAMES 2
ASSERT scene==0

# ---- Drive the column to 0 so UP lands on chip 0 (column-matched nav) ----
LEFT
FRAMES 1
LEFT
FRAMES 1
LEFT
FRAMES 1
LEFT
FRAMES 1
LEFT
FRAMES 1

# ---- From the arranger grid, UP selects chip 0 (via the chip-row wrapper) ----
UP
# 4 frames: tolerate a cold-start frame where the first reflow hasn't run
FRAMES 4
ASSERT selected==chip

# ---- Expand chip 0 with EDIT+UP ----
EDIT UP
FRAMES 3
ASSERT chipExpanded==0 1

# ---- Cycle the swatch (RIGHT cycles colour while expanded, no edit) ----
RIGHT
FRAMES 3
ASSERT chipColour==0 1
RIGHT
FRAMES 3
ASSERT chipColour==0 2

# ---- Label edit: cycle char at cursor (EDIT+UP), advance cursor (EDIT+RIGHT) ----
# The chip-label buffer is 8 chars + NUL terminator (CHIP_LABEL_MAX);
# trailing positions stay at NUL so the C-string shortens as we go.
# chipLabelCharIndex maps NUL -> 0, so the first EDIT UP lands on
# NAME_CHARS[1] = 'B' (see chipLabelCycleChar in src/gui.c).
EDIT UP
FRAMES 3
ASSERT chipLabel==0 "B"
EDIT RIGHT
FRAMES 3
EDIT UP
FRAMES 3
ASSERT chipLabel==0 "BB"
EDIT RIGHT
FRAMES 3
EDIT UP
FRAMES 3
ASSERT chipLabel==0 "BBB"

# ---- Collapse ----
EDIT
FRAMES 3
ASSERT chipExpanded==0 0

# ---- Jump to instrument page (EDIT + RIGHT on collapsed chip) ----
EDIT RIGHT
FRAMES 5
ASSERT scene==1

# ---- From the instrument page, exercise the meta row ----
# Selection starts on RATIO1 (the first FM control) after chip-jump;
# navigate UP to the closest preset-row button so we can confirm the
# harness round-trips into the FM controls cleanly.
UP
FRAMES 3
ASSERT selected==NEXT
DOWN
FRAMES 3
ASSERT selected==RATIO1
# ---- Land on the TNEXT action button (deterministically via JUMP by name) ----
# The meta row is [TPREV] [TYPE-tag label (non-selectable)] [TNEXT] [VOICES]
# [WIDTH]. JUMP VOICES + LEFT skips VOICES -> TNEXT (the type-cycle button,
# not the old single TYPE button — Task 5 rework removed that). Bare
# UP/LEFT navigation from RATIO1 is fragile across bestInDirection
# interpretations, so use the JUMP escape hatch.
JUMP VOICES
FRAMES 1
ASSERT selected==VOICES
# LEFT in btnwrap (horizontal) takes us leftward from VOICES to TNEXT.
LEFT
FRAMES 3
# Cycle voice type: FM -> BLEP (TNEXT fires cbTypeNext)
ASSERT voiceType==FM

EDIT
FRAMES 3
ASSERT voiceType==BLEP
# After the cycle, the TYPE tag label is rebuilt (drawTypeLabelGuiNode
# re-renders with the new voice-type tag). The TNEXT button itself is
# stable, but to be safe we re-JUMP VOICES + LEFT to land deterministically.
JUMP VOICES
FRAMES 1
LEFT
FRAMES 3
# BLEP -> SAMPLE.
EDIT
FRAMES 3
ASSERT voiceType==SAMPLE

# ---- VOICES dial: drive the adjust callback directly ----
JUMP VOICES
FRAMES 1
ASSERT selected==VOICES
# EDIT RIGHT increments the dial (+0.1 per press; voice counts are
# integers, so each press bumps the count by 1).
EDIT RIGHT
FRAMES 3
ASSERT voiceCount==2
EDIT RIGHT
FRAMES 3
ASSERT voiceCount==3
EDIT RIGHT
FRAMES 3
ASSERT voiceCount==4

QUIT
