# env_stage.txt - envelope stage box: rate/curve nav + curve edit (Task 8)
#
# The envelope's stage dials live on the MODULATION SCREEN's detail pane
# now -- the instrument screen's modulation list is gone, and the screen
# that shows a source's own parameters is where they are turned.
#
# Default boot: SCENE_INSTRUMENT, selection = RATIO1 (FM channel 0).
# Each FM channel has envelope mods; env[0] is the built-in ADSR-ish
# envelope. Each stage renders as a box with a rate dial (ATK / DEC) and a
# small curve dial (CRV) whose value slot paints the curvature glyph.
FRAMES 4

# --- Open the modulation screen -------------------------------------------
# The line is the instrument screen's entry point into it.
JUMP MODLINE
EDIT
FRAMES 2
ASSERT topLayerSelected==MODSRC_0

# --- Deterministic entry: the envelope's stage 0 rate dial ----------------
# JUMPTL selects in the top layer's graph -- the screen's own.
JUMPTL ATK
ASSERT topLayerSelected==ATK

# Stage 0 box: ATK (rate) -> CRV (curve) are siblings, so RIGHT walks
# from the rate dial into the curve dial.
RIGHT
ASSERT topLayerSelected==CRV
ASSERT envCurvature0f==0.75

# EDIT+RIGHT nudges the curvature; the glyph index and the param move.
# A layer's dial turns with EDIT+arrow, exactly as on the instrument
# screen.
EDIT RIGHT
ASSERT envCurvature0f!=0.75

# Stage 1 (decay) is a separate box; its curvature is independent of the
# stage 0 edit.
JUMPTL DEC
ASSERT topLayerSelected==DEC
RIGHT
ASSERT topLayerSelected==CRV
ASSERT envCurvature1f==0.75
QUIT
