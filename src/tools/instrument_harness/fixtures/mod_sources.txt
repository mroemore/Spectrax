# mod_sources.txt - the modulation screen's source list end-to-end.
#
# The instrument screen's modulation list is gone: the modulation screen
# carries the full interface now. This fixture drives the source list
# there -- ADD a runtime source, cycle its type (ENV -> LFO -> RND ->
# ENV), route it into a destination, unroute it, and delete it.
#
# FM instrument: core 4 envelopes, then runtime sources, then the four
# pattern tracks.

FRAMES 4

# --- Open the modulation screen -------------------------------------------
JUMP MODLINE
EDIT
FRAMES 2
ASSERT topLayerSelected==MODSRC_0

# --- ADD a runtime source -------------------------------------------------
JUMPTL MODLIST_ADD
FRAMES 1
EDIT
FRAMES 2
# The boot patch has 4 core envelopes, 1 LFO and 4 pattern tracks, so the
# added source is the ninth (position 8).
ASSERT envcount==5
ASSERT topLayerSelected==MODSRC_8

# --- 3-way type cycle on the new source (ENV -> LFO -> RND -> ENV) --------
# The detail pane carries the TYPE control for the source the screen is
# showing.
JUMPTL MOD_TYPE
FRAMES 1
EDIT
FRAMES 2
# The rebuild resets the screen's selection, so it is re-selected by
# name. The type is now LFO: its own parameters are RATE and SHAPE.
JUMPTL MOD_TYPE
FRAMES 1
EDIT
FRAMES 2
JUMPTL MOD_TYPE
FRAMES 1
EDIT
FRAMES 2
JUMPTL MOD_TYPE
FRAMES 1
EDIT
FRAMES 2
# RND wraps back to ENV.
JUMPTL MOD_TYPE
FRAMES 1

# --- Route the source to a destination via the picker ----------------------
# "+ ROUTE" opens the plain destination list.
JUMPTL MOD_ADDRUTE
FRAMES 1
EDIT
FRAMES 2
ASSERT topLayerSelected==DEST_0
EDIT
FRAMES 2
ASSERT topLayerSelected==MODSRC_8

# --- Delete via the confirm modal -----------------------------------------
JUMPTL MODLIST_DEL
FRAMES 1
EDIT
FRAMES 2
ASSERT topLayerSelected==DELETE_NO
# EDIT on NO cancels: the source survives.
EDIT
FRAMES 2
ASSERT envcount==5
# Re-open + confirm YES: the source goes.
JUMPTL MODLIST_DEL
FRAMES 1
EDIT
FRAMES 2
ASSERT topLayerSelected==DELETE_NO
RIGHT
ASSERT topLayerSelected==DELETE_YES
EDIT
FRAMES 2
ASSERT envcount==4

QUIT
