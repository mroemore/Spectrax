# add_route_delete.txt
# The instrument screen's modulation LIST is gone: the modulation screen
# carries the full interface now. This fixture drives the same flow
# there -- ADD a source, route it, unroute it, cycle its type, and
# delete it (both the cancel and the confirm path).
#
# The screen is reached from the instrument screen's modulation LINE,
# which is the screen's only entry point.

FRAMES 1
ASSERT selected==RATIO1
# The boot preset (saved fm1.ipb) still has 4 AD envelopes.
ASSERT envcount==4

# --- Open the modulation screen -------------------------------------------
JUMP MODLINE
EDIT
FRAMES 2
ASSERT topLayerSelected==MODSRC_0

# --- ADD a runtime source -------------------------------------------------
# The source list's header carries the ADD button.
JUMPTL MODLIST_ADD
FRAMES 1
ASSERT topLayerSelected==MODLIST_ADD
EDIT
FRAMES 2
# ADD adds a runtime envelope and the screen follows it.
ASSERT envcount==5

# --- Route it --------------------------------------------------------------
# "+ ROUTE" opens the destination picker: a plain list of the
# instrument's routable parameters.
JUMPTL MOD_ADDRUTE
FRAMES 1
ASSERT topLayerSelected==MOD_ADDRUTE
EDIT
FRAMES 2
ASSERT topLayerSelected==DEST_0
# Pick the first destination: the source is routed into it.
EDIT
FRAMES 2
# The picker is gone and the screen's route list came back rebuilt.
ASSERT topLayerSelected==MODSRC_8

# --- Unroute it ------------------------------------------------------------
# The route list holds a row per route; the picker is opened from
# "+ ROUTE" again, and the destination list offers the same names.
JUMPTL MOD_ADDRUTE
FRAMES 1
EDIT
FRAMES 2
ASSERT topLayerSelected==DEST_0
EDIT
FRAMES 2
# The route is a toggle: picking a routed destination removes it.
ASSERT topLayerSelected==MODSRC_8

# --- Cycle the source's type ----------------------------------------------
# The detail pane carries the TYPE control for the source the screen is
# showing.
JUMPTL MOD_TYPE
FRAMES 1
ASSERT topLayerSelected==MOD_TYPE
EDIT
FRAMES 2
# ENV -> LFO. The rebuild resets the selection, so the screen is
# re-selected by name.
JUMPTL MOD_TYPE
FRAMES 1
EDIT
FRAMES 2

# --- Delete it: the cancel path -------------------------------------------
JUMPTL MODLIST_DEL
FRAMES 1
EDIT
FRAMES 2
ASSERT topLayerSelected==DELETE_NO
# EDIT on DELETE_NO cancels: the source survives.
EDIT
FRAMES 2
ASSERT envcount==5

# --- Delete it: the confirm path ------------------------------------------
JUMPTL MODLIST_DEL
FRAMES 1
EDIT
FRAMES 2
ASSERT topLayerSelected==DELETE_NO
RIGHT
ASSERT topLayerSelected==DELETE_YES
EDIT
FRAMES 2
ASSERT envcount==4

QUIT
