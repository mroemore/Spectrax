# pagenav.txt - page navigation on the instrument screen.
# FUNCTION+UP/DOWN: vertical page nav across the 4 groups (META, preset,
# FM controls, modulators). FUNCTION+LEFT/RIGHT: horizontal row-edge nav.
# SELECT+UP/DOWN = channel cycling; SELECT+LEFT/RIGHT = scene switch.

FRAMES 4
ASSERT selected==RATIO1

# FUNCTION+DOWN -> FM group's lowest (ALG, the bottom row of the FM box).
HOLD FUNCTION
DOWN
RELEASE
FRAMES 2
ASSERT selected==ALG

# FUNCTION+DOWN again -> wrap to the modulators group's top (MODS_ADD).
HOLD FUNCTION
DOWN
RELEASE
FRAMES 2
ASSERT selected==MODS_ADD

# FUNCTION+DOWN again -> modulators group's lowest (env3's ATTACK).
HOLD FUNCTION
DOWN
RELEASE
FRAMES 2
ASSERT selected==ATTACK

# FUNCTION+UP -> back to the modulators top.
HOLD FUNCTION
UP
RELEASE
FRAMES 2
ASSERT selected==MODS_ADD

# FUNCTION+UP again -> wrap to the previous group's lowest (FM bottom = ALG).
HOLD FUNCTION
UP
RELEASE
FRAMES 2
ASSERT selected==ALG

# FUNCTION+RIGHT -> row's rightmost (GAIN, the last dial in the FM bottom row).
HOLD FUNCTION
RIGHT
RELEASE
FRAMES 2
ASSERT selected==GAIN

# FUNCTION+LEFT -> back to the row's leftmost (ALG).
HOLD FUNCTION
LEFT
RELEASE
FRAMES 2
ASSERT selected==ALG

QUIT
