# pagenav.txt - page navigation on the instrument screen.
# FUNCTION+UP/DOWN: vertical page nav across the 4 groups (META, preset,
# the voice's controls, the modulators). FUNCTION+LEFT/RIGHT: horizontal
# row-edge nav.
# SELECT+UP/DOWN = channel cycling; SELECT+LEFT/RIGHT = scene switch.
#
# The modulators group is now the modulation LINE alone: the modulation
# list is gone from the instrument screen, and the line is what stands in
# its place.

FRAMES 4
ASSERT selected==RATIO1

# FUNCTION+DOWN -> the voice controls group's lowest (ALG, the bottom row
# of the FM box).
HOLD FUNCTION
DOWN
RELEASE
FRAMES 2
ASSERT selected==ALG

# FUNCTION+DOWN again -> wrap to the modulators group's top (MODLINE).
HOLD FUNCTION
DOWN
RELEASE
FRAMES 2
ASSERT selected==MODLINE

# FUNCTION+UP -> back to the voice controls group's lowest (ALG).
HOLD FUNCTION
UP
RELEASE
FRAMES 2
ASSERT selected==ALG

# FUNCTION+RIGHT -> row's rightmost (GAIN, the last dial in the FM bottom row).
HOLD FUNCTION
RIGHT
RELEASE
FRAMES 2
ASSERT selected==GAIN

# FUNCTION+LEFT -> back to the row's leftmost (ALG).
HOLD FUNCTION
LEFT
RELEASE
FRAMES 2
ASSERT selected==ALG

QUIT
